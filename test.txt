Автоматическое тестирование: Что и как прогнать
Что автоматизируется
Уровень 1: Юнит-тесты чистых функций        → Vitest (уже есть)
Уровень 2: Интеграционные тесты сервисов    → Vitest + Firebase Emulator
Уровень 3: Комбинаторика движка             → Генератор тестовых случаев
Уровень 4: Пользовательские сценарии        → Генератор сценариев
Уровень 5: Валидация данных                 → Свойственное тестирование
Уровень 6: E2E (UI)                         → Playwright (опционально)

// core/inference/__tests__/dominant.test.ts

import { describe, it, expect } from 'vitest';
import { getDailyDominant } from '../dominant';

describe('getDailyDominant', () => {
  it('определяет доминанту из текущего вектора, не из профиля', () => {
    const currentVector = { anger: 0.8, joy: 0.2 } as PlutchikVector;
    const profile = { joy: 0.6, trust: 0.4 } as PlutchikVector;
    
    const dominant = getDailyDominant(currentVector, [], profile);
    
    // Доминанта должна быть Гнев (из текущего), не Радость (из профиля)
    expect(dominant).toBe('anger');
  });
  
  it('пульсы корректируют доминанту', () => {
    const currentVector = { joy: 0.6 } as PlutchikVector;
    const pulses = [
      { fear: 0.9 } as PlutchikVector,
      { fear: 0.8 } as PlutchikVector,
    ];
    const profile = { joy: 0.6 } as PlutchikVector;
    
    const dominant = getDailyDominant(currentVector, pulses, profile);
    
    // Пульсы страха перевешивают утреннюю радость
    expect(dominant).toBe('fear');
  });
  
  it('тай-брейкер: при равных значениях смотрит профиль', () => {
    const currentVector = { joy: 0.5, trust: 0.5 } as PlutchikVector;
    const profile = { trust: 0.7, joy: 0.3 } as PlutchikVector;
    
    const dominant = getDailyDominant(currentVector, [], profile);
    
    // Разница < 0.05 → смотрим профиль → доверие ближе
    expect(dominant).toBe('trust');
  });
});

Вместо того чтобы писать тесты для каждой комбинации вручную, генерируем их:

// core/__tests__/engineCombinatorics.test.ts

import { describe, it, expect } from 'vitest';
import { runRecommendationEngine } from '../recommendation/engine';

const EMOTIONS: EmotionKey[] = ['joy', 'trust', 'fear', 'surprise', 'sadness', 'disgust', 'anger', 'anticipation'];
const AROUSAL_LEVELS = ['low', 'medium', 'high'] as const;
const TIME_OF_DAY = ['morning', 'afternoon', 'evening'] as const;

describe('Детерминированный движок — комбинаторика', () => {
  // Генерируем все комбинации: 8 эмоций × 3 уровня × 3 времени = 72 комбинации
  EMOTIONS.forEach(emotion => {
    AROUSAL_LEVELS.forEach(arousal => {
      TIME_OF_DAY.forEach(timeOfDay => {
        it(`${emotion} / ${arousal} / ${timeOfDay} → валидная рекомендация`, () => {
          const vector = createVectorWithDominant(emotion, getIntensity(arousal));
          const context = createContext({ timeOfDay, arousal });
          
          const result = runRecommendationEngine(vector, context);
          
          // Базовые проверки
          expect(result.practice).toBeDefined();
          expect(result.oil).toBeDefined();
          expect(result.color).toMatch(/^#[0-9A-Fa-f]{6}$/);
          
          // Практика должна быть из списка
          expect(ALL_PRACTICES).toContain(result.practice);
          
          // Масло не должно быть забанено
          expect(context.bannedOils).not.toContain(result.oil);
          
          // Цвет должен соответствовать эмоции
          expect(result.color).toBe(getColorForEmotion(emotion));
        });
      });
    });
  });
});

describe('Детерминированный движок — граничные случаи', () => {

// core/__tests__/propertyBased.test.ts

import fc from 'fast-check';
import { describe, it, expect } from 'vitest';
import { inferPulseState } from '../inference/pulseInference';

describe('Свойственное тестирование: микро-ввод', () => {
  it('любой текст не вызывает падение', () => {
    fc.assert(
      fc.property(fc.string(), (text) => {
        const result = inferPulseState({ text });
        
        // Результат всегда должен быть валидным
        expect(result.vector).toBeDefined();
        expect(result.confidence).toBeGreaterThanOrEqual(0);
        expect(result.confidence).toBeLessThanOrEqual(1);
        
        // Вектор всегда в пределах [0, 1]
        Object.values(result.vector).forEach(v => {
          expect(v).toBeGreaterThanOrEqual(0);
          expect(v).toBeLessThanOrEqual(1);
        });
      }),
      { numRuns: 1000 }  // 1000 случайных строк
    );
  });
  
  it('любой вектор Плутчика → валидная доминанта', () => {
    const vectorArbitrary = fc.record({
      joy: fc.float({ min: 0, max: 1 }),
      trust: fc.float({ min: 0, max: 1 }),
      fear: fc.float({ min: 0, max: 1 }),
      surprise: fc.float({ min: 0, max: 1 }),
      sadness: fc.float({ min: 0, max: 1 }),
      disgust: fc.float({ min: 0, max: 1 }),
      anger: fc.float({ min: 0, max: 1 }),
      anticipation: fc.float({ min: 0, max: 1 }),
    });
    
    fc.assert(
      fc.property(vectorArbitrary, (vector) => {
        const dominant = getDailyDominant(vector as PlutchikVector, [], vector as PlutchikVector);
        
        expect(EMOTIONS).toContain(dominant);
      }),
      { numRuns: 5000 }  // 5000 случайных векторов
    );
  });
});

import { describe, it, expect, beforeAll, afterAll } from 'vitest';

      source: 'local' as const,

  });
  
  it('дубль пульса не добавляется', async () => {
    const service = new PulseSyncService(db);
    
    const pulse = { id: 'pulse-1', /* ... */ };
    
    await service.savePulseLocal(testDate, pulse);
    await service.savePulseLocal(testDate, pulse); // дубль
    
    const doc = await getDoc(doc(db, 'emotionalGraph', `${testUid}_${testDate}`));
    expect(doc.data().pulses).toHaveLength(1); // не 2

Хелпер для симуляции

// e2e/helpers/simulateUserJourney.ts

interface JourneyConfig {
  days: number;
  skipDays?: number[];
  offlineDays?: number[];
  emotions?: 'random' | 'fixed' | EmotionKey;
  pulses?: Array<{ day: number; vector: PlutchikVector; decision: string }>;
  emotionPattern?: { morning: PlutchikVector; evening: PlutchikVector };
  calibrationAnswers?: Record<string, Answer[]>;
  applyPattern?: boolean;
}

async function simulateUserJourney(config: JourneyConfig): Promise<JourneyResult> {
  const result: JourneyResult = {
    days: [],
    profile: createInitialProfile(),
    unlockedFeatures: [],
    detectedPatterns: [],
    calibrations: [],
    localStorage: { days: {} },
    firestore: { days: {} },
  };
  
  for (let day = 1; day <= config.days; day++) {
    // Пропуск дня
    if (config.skipDays?.includes(day)) {
      result.days.push({ day, status: 'skipped' });
      continue;
    }
    
    // Офлайн
    const isOffline = config.offlineDays?.includes(day) ?? false;
    
    // Утренний ритуал
    const morningVector = getMorningVector(config, day);
    const morningResult = runRecommendationEngine(morningVector, createContext(day));
    
    // Пульсы
    const dayPulses = config.pulses?.filter(p => p.day === day) ?? [];
    
    // Вечерний фидбек
    const feedback = getRandomFeedback();
    
    // Калибровка
    if (config.calibrationAnswers?.[`day${day}`]) {
      const answers = config.calibrationAnswers[`day${day}`];
      result.profile = applyCalibration(result.profile, answers);
      result.calibrations.push({ day, answers });
    }
    
    // Разблокировки
    const unlocked = checkUnlocks(day, result.longestStreak);
    result.unlockedFeatures.push(...unlocked);
    
    // Паттерны
    if (day >= 14) {
      const patterns = detectPatterns(result.days);
      result.detectedPatterns.push(...patterns);
    }
    
    result.days.push({
      day,
      morningVector,
      morningResult,
      pulses: dayPulses,
      feedback,
      isOffline,
      status: 'completed',
    });
  }
  
  return result;
}
